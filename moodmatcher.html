document.getElementById('find-song-btn').addEventListener('click', function () { // Initialize scores for each genre let scores = { "Banger!": 0, "Happy!": 0, "Ballad!": 0, "Duet!": 0, "Soulful!": 0 }; // Get the form and ensure all questions are answered let form = document.getElementById('quiz-form'); let allAnswered = true; // Loop through each question and update scores for (let i = 1; i <= 5; i++) { let selectedOption = form['q' + i].value; if (!selectedOption) { allAnswered = false; alert("Please answer all questions!"); return; // Exit if any question is unanswered } scores[selectedOption]++; // Increment the score for the selected genre } // Determine the genre with the highest score let highestGenre = Object.keys(scores).reduce((a, b) => scores[a] > scores[b] ? a : b); // Fetch the CSV data and filter songs based on the highest genre fetch('data.csv') .then(response => response.text()) .then(csv => { // Parse the CSV data const rows = csv.trim().split('\n'); let filteredSongs = []; // Loop through the rows and filter songs by the highest genre for (let i = 1; i < rows.length; i++) { let row = rows[i].split(','); let songTitle = row[0].trim(); let songArtist = row[1].trim(); let songType = row[2].trim(); if (songType === highestGenre) { filteredSongs.push({ Artist: songArtist, Title: songTitle, Type: songType }); } } // If songs are found, randomly select one and display it if (filteredSongs.length > 0) { let randomSong = filteredSongs[Math.floor(Math.random() * filteredSongs.length)]; document.getElementById('song-result').innerHTML = `

Your Perfect Karaoke Song:

${randomSong.Artist} - ${randomSong.Title}

Genre: ${randomSong.Type}

`; } else { // Fallback message if no songs are found document.getElementById('song-result').innerHTML = `

No songs found for your mood. Try again!

`; } }) .catch(error => { console.error('Error fetching or processing the CSV file:', error); document.getElementById('song-result').innerHTML = `

Oops! Something went wrong. Please try again later.

`; }); });
